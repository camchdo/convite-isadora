    <script>

        const yesButton = document.getElementById("yes");
        const noButton = document.getElementById("no");

        const invite = document.getElementById("invite");
        const success = document.getElementById("success");


        /*
         * POSIÇÃO INICIAL
         *
         * O botão começa ao lado do "EU VOU".
         */

        function initialPosition() {

            const yesRect = yesButton.getBoundingClientRect();

            noButton.style.left =
                (yesRect.right + 12) + "px";

            noButton.style.top =
                (yesRect.top) + "px";
        }

        initialPosition();


        /*
         * MOVE O BOTÃO
         *
         * Ele sempre calcula uma posição que:
         *
         * - esteja dentro da tela;
         * - não corte o botão;
         * - possa ser clicada;
         * - seja diferente da posição anterior.
         */

        function moveNoButton() {

            const buttonWidth = noButton.offsetWidth;
            const buttonHeight = noButton.offsetHeight;

            const padding = 10;

            const maxX =
                window.innerWidth - buttonWidth - padding;

            const maxY =
                window.innerHeight - buttonHeight - padding;

            let newX;
            let newY;

            do {

                newX =
                    Math.random() * (maxX - padding) + padding;

                newY =
                    Math.random() * (maxY - padding) + padding;

            } while (
                Math.abs(newX - noButton.offsetLeft) < 100 &&
                Math.abs(newY - noButton.offsetTop) < 70
            );

            noButton.style.left = newX + "px";
            noButton.style.top = newY + "px";
        }


        /*
         * Quando a pessoa tenta clicar,
         * o botão simplesmente foge.
         */

        noButton.addEventListener("click", function(event) {

            event.preventDefault();

            moveNoButton();

        });


        /*
         * Também foge quando o mouse chega perto.
         *
         * Isso torna praticamente impossível clicar
         * no botão pelo computador.
         */

        noButton.addEventListener("mouseenter", function() {

            moveNoButton();

        });


        /*
         * No celular, o toque também faz o botão fugir.
         */

        noButton.addEventListener("touchstart", function(event) {

            event.preventDefault();

            moveNoButton();

        });


        /*
         * QUANDO CLICAR EM "EU VOU"
         */

        yesButton.addEventListener("click", function() {

            invite.style.display = "none";
            noButton.style.display = "none";

            success.style.display = "flex";

        });


        /*
         * Se a tela mudar de tamanho,
         * reposiciona o botão.
         */

        window.addEventListener("resize", function() {

            if (noButton.style.display !== "none") {
                moveNoButton();
            }

        });

    </script>
